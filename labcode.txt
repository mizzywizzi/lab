#include <stdio.h>
#include <string.h>

int flag = 0, len, pt = 0;
char st[25];

void e();
void t();
void f();
void eprime();
void tprime();
void advance();

int main()
{
    printf("\nRecursive Descent Parser\n");

    printf("\nGrammar:\n");
    printf("E  -> TE'\n");
    printf("E' -> +TE' | epsilon\n");
    printf("T  -> FT'\n");
    printf("T' -> *FT' | epsilon\n");
    printf("F  -> (E) | a\n");

    printf("\nEnter the string: ");
    scanf("%24s", st);

    len = strlen(st);

    e();

    if(flag == 0 && pt == len)
        printf("\nString is Accepted\n");
    else
        printf("\nString is Rejected\n");

    return 0;
}

void e()
{
    t();
    eprime();
}

void t()
{
    f();
    tprime();
}

void eprime()
{
    if(st[pt] == '+')
    {
        advance();
        t();
        eprime();
    }
}

void tprime()
{
    if(st[pt] == '*')
    {
        advance();
        f();
        tprime();
    }
}

void f()
{
    if(st[pt] == '(')
    {
        advance();
        e();

        if(st[pt] == ')')
            advance();
        else
            flag = 1;
    }
    else if(st[pt] == 'a')
    {
        advance();
    }
    else
    {
        flag = 1;
    }
}

void advance()
{
    if(st[pt] != '\0')
        pt++;
}

#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main(int argc, char *argv[])
{
    FILE *f1, *f2;
    int c;
    char str[100];
    int lineno = 1, num = 0, i = 0;

    if (argc < 2)
    {
        printf("Usage: %s <input_file>\n", argv[0]);
        return 1;
    }

    f1 = fopen(argv[1], "r");

    if (f1 == NULL)
    {
        printf("Error: Could not open input file %s\n", argv[1]);
        return 1;
    }

    f2 = fopen("output.txt", "w");

    if (f2 == NULL)
    {
        printf("Error: Could not create output.txt\n");
        fclose(f1);
        return 1;
    }

    while ((c = getc(f1)) != EOF)
    {
        if (isdigit(c))
        {
            num = c - '0';
            c = getc(f1);

            while (isdigit(c))
            {
                num = num * 10 + (c - '0');
                c = getc(f1);
            }

            fprintf(f2, "%d is a number\n", num);
            ungetc(c, f1);
        }

        else if (isalpha(c))
        {
            i = 0;
            str[i++] = (char)c;
            c = getc(f1);

            while (isdigit(c) || isalpha(c) ||
                   c == '_' || c == '$')
            {
                if (i < 99)
                    str[i++] = (char)c;

                c = getc(f1);
            }

            str[i] = '\0';

            if (strcmp("for", str) == 0 ||
                strcmp("while", str) == 0 ||
                strcmp("do", str) == 0 ||
                strcmp("int", str) == 0 ||
                strcmp("float", str) == 0 ||
                strcmp("char", str) == 0 ||
                strcmp("double", str) == 0 ||
                strcmp("static", str) == 0 ||
                strcmp("case", str) == 0)
            {
                fprintf(f2, "%s is a keyword\n", str);
            }
            else
            {
                fprintf(f2, "%s is an identifier\n", str);
            }

            ungetc(c, f1);
        }

        else if (c == ' ' || c == '\t')
        {
            fprintf(f2, "\n");
        }

        else if (c == '\n')
        {
            lineno++;
        }

        else
        {
            fprintf(f2, "%c is a special symbol\n", c);
        }
    }

    printf("Total no of lines are: %d\n", lineno);

    fclose(f1);
    fclose(f2);

    return 0;
}

#include <stdio.h>
#include <string.h>

int main()
{
    char icode[20][30], str[30], opr[10];
    int i = 0;

    printf("\nEnter the set of intermediate code (terminated by exit):\n");

    // Input intermediate code lines
    do
    {
        scanf("%s", icode[i]);
    } while (strcmp(icode[i++], "exit") != 0);

    printf("\nTarget Code Generation");
    printf("\n******************************");

    i = 0;

    // Process each intermediate code line
    do
    {
        strcpy(str, icode[i]);

        // Check if it's an assignment-only statement (like a=b)
        if (strlen(str) == 3)
        {
            printf("\n\tMov %c,R%d", str[2], i);
            printf("\n\tMov R%d,%c", i, str[0]);
        }
        else
        {
            // Map operator to assembly mnemonic
            switch (str[3])
            {
                case '+':
                    strcpy(opr, "ADD");
                    break;
                case '-':
                    strcpy(opr, "SUB");
                    break;
                case '*':
                    strcpy(opr, "MUL");
                    break;
                case '/':
                    strcpy(opr, "DIV");
                    break;
                default:
                    strcpy(opr, "???"); // Unknown operator
            }


            // Generate target code
            printf("\n\tMov %c,R%d", str[2], i);
            printf("\n\t%s %c,R%d", opr, str[4], i);
            printf("\n\tMov R%d,%c", i, str[0]);
        }

    } while (strcmp(icode[++i], "exit") != 0);

    return 0;
}
